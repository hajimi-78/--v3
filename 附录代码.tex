def thomas(a,b,c,d):
    n=len(b); cp=np.zeros(n); dp=np.zeros(n)
    cp[0]=c[0]/b[0]; dp[0]=d[0]/b[0]
    for i in range(1,n):
        m=b[i]-a[i]*cp[i-1]; cp[i]=c[i]/m; dp[i]=(d[i]-a[i]*dp[i-1])/m
    x=np.zeros(n); x[-1]=dp[-1]

    return x
